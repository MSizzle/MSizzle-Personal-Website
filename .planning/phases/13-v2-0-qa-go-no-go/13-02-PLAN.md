Run 3 Lighthouse desktop runs per route. Use the exact invocation (substitute PREVIEW_URL and FIRST_POST_SLUG):

```
PREVIEW_URL=<captured-from-task-1>
FIRST_POST_SLUG=<first-post-slug>
EVIDENCE_DIR=".planning/phases/13-v2-0-qa-go-no-go/evidence/lighthouse"

for route_name in home about prometheus blog-index blog-post; do
  case $route_name in
    home)       url="$PREVIEW_URL/" ;;
    about)      url="$PREVIEW_URL/about" ;;
    prometheus) url="$PREVIEW_URL/prometheus" ;;
    blog-index) url="$PREVIEW_URL/blog" ;;
    blog-post)  url="$PREVIEW_URL/blog/$FIRST_POST_SLUG" ;;
  esac
  for run in 1 2 3; do
    npx lighthouse "$url" \
      --preset=desktop \
      --output=json \
      --output-path="$EVIDENCE_DIR/$route_name/run-$run.json" \
      --chrome-flags="--headless --no-sandbox" \
      --quiet
    echo "Done: $route_name run $run"
  done
done
```

After all 15 runs complete, extract the four key scores from each JSON file using node (or jq if available):

```
node -e "
const fs = require('fs');
const routes = ['home','about','prometheus','blog-index','blog-post'];
routes.forEach(r => {
  const runs = [1,2,3].map(i => {
    const data = JSON.parse(fs.readFileSync('.planning/phases/13-v2-0-qa-go-no-go/evidence/lighthouse/' + r + '/run-' + i + '.json', 'utf8'));
    return {
      perf: Math.round(data.categories.performance.score * 100),
      a11y: Math.round(data.categories.accessibility.score * 100),
      bp: Math.round(data.categories['best-practices'].score * 100),
      seo: Math.round(data.categories.seo.score * 100)
    };
  });
  const median = arr => arr.sort((a,b)=>a-b)[1];
  console.log(r + ' | P:' + [runs[0].perf,runs[1].perf,runs[2].perf].join(',') + ' -> median:' + median([runs[0].perf,runs[1].perf,runs[2].perf]));
  console.log(r + ' | A:' + [runs[0].a11y,runs[1].a11y,runs[2].a11y].join(',') + ' -> median:' + median([runs[0].a11y,runs[1].a11y,runs[2].a11y]));
  console.log(r + ' | BP:' + [runs[0].bp,runs[1].bp,runs[2].bp].join(',') + ' -> median:' + median([runs[0].bp,runs[1].bp,runs[2].bp]));
  console.log(r + ' | SEO:' + [runs[0].seo,runs[1].seo,runs[2].seo].join(',') + ' -> median:' + median([runs[0].seo,runs[1].seo,runs[2].seo]));
});
"
```
