  export function ListRow({ title, href, extra, meta, big = false, last = false }: Props) {
    return (
      <div className={cn(!last && 'border-b border-rule')}>
        <Link
          href={href}
          className={cn(
            'grid grid-cols-[1fr_auto] items-baseline gap-6 text-ink no-underline',
            big ? 'py-7' : 'py-5'
          )}
        >
          <div>
            <div className={cn(big ? 'text-list-title' : 'text-list-title-home')}>
              {title}
            </div>
            {extra && (
              <div className="mt-1.5 text-caption text-muted">{extra}</div>
            )}
          </div>
          {meta && (
            <div className="whitespace-nowrap text-meta uppercase text-muted">{meta}</div>
          )}
        </Link>
      </div>
    )
  }

CRITICAL: grid-cols-[1fr_auto] IS a Tailwind v4 arbitrary value, but it is the standard Tailwind grid-template-columns shorthand and is documented as the idiomatic v4 pattern (not a token-replaceable value). The D-09 rule against arbitrary values targets color/size/tracking tokens, not grid templates. Acceptance criteria checks for ARBITRARY VALUE patterns in color/size/spacing positions (border-[, bg-[, text-[Npx], tracking-[) but ALLOWS grid-cols-[ as the canonical pattern.

Why `text-caption` instead of `text-[13px]`: Plan 09-01 introduced `--text-caption: 13px` specifically because the handoff says the extra blurb is 13–15px (D-14a). The other option (`text-meta` at 11px) is too small. Use `text-caption`.

Why `mt-1.5` (6px) is OK: Tailwind's standard spacing scale. `mt-1.5` is `margin-top: 0.375rem` (6px) — a token-scale increment, not arbitrary like `mt-[6px]`.
